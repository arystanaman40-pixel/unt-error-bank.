  function copyText(text){

    if (navigator.clipboard && navigator.clipboard.writeText) {

    } else fallbackCopy(text);

    ta.value = text; ta.style.position = "fixed"; ta.style.opacity = "0";

  function apiGet(action, params){

    var url = getServerUrl();

    if (!url) return Promise.reject(new Error("Server not configured yet"));

    var qs = Object.keys(params || {}).map(function(k){ return encodeURIComponent(k) + "=" + encodeURIComponent(params[k]); }).join("&");

    return fetch(url + "?action=" + action + (qs ? "&" + qs : ""))

      .then(function(r){ return r.json(); })

      .then(function(data){ if (!data.ok) throw new Error(data.error || "Server error"); return data; });

  }

  function apiPost(action, body){

    var url = getServerUrl();

    if (!url) return Promise.reject(new Error("Server not configured yet"));

    var payload = Object.assign({ action: action }, body || {});

    return fetch(url, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(payload) })

  function currentlyOnQuiz(testId){ return studentSession && studentSession.testId === testId && studentSession.step === "quiz"; }


  function renderStudentQuiz(){

    var s = studentSession, t = s.test, qi = s.current, q = t.questions[qi];

    var L = letters(), isMulti = q.type === "multiple", chosen = s.answers[q.id];


    var dotsHtml = t.questions.map(function(qq, i){

      var answered = QUESTION_TYPES[qq.type].isAnswered(s.answers[qq.id]);

      return '<div class="dot' + (answered ? " answered" : "") + (i === qi ? " current" : "") + '" data-jump="' + i + '">' + (i+1) + '</div>';

    }).join("");

    var optsHtml = q.options.map(function(opt, i){

      var sel = isMulti ? chosen.indexOf(i) !== -1 : chosen === i;

      return '<button class="option' + (sel ? " selected" : "") + '" data-opt="' + i + '">' +

        '<span class="letter">' + L[i] + '</span><span>' + escapeHtml(opt) + '</span></button>';

    }).join("");

    var answeredCount = t.questions.filter(function(qq){ return QUESTION_TYPES[qq.type].isAnswered(s.answers[qq.id]); }).length;


    var statusMap = {

      idle: "", saving: '<span style="color:var(--gold);">⟳ Saving…</span>',

      saved: '<span style="color:var(--correct);">✓ Saved</span>',

      offline: '<span style="color:var(--wrong);">⚠ Offline — saved on this device</span>'

    };

    var errBanner = s.submitError ? '<p class="small mt8" style="color:var(--wrong);">' + escapeHtml(s.submitError) + '</p>' : "";


    app.innerHTML = '<div class="wrap narrow">' + topbarHtml(true) +

      '<div class="exam-card">' +

        '<div class="q-meta"><span class="q-count">QUESTION ' + (qi+1) + ' / ' + t.questions.length + '</span>' +

        '<span class="small">' + (statusMap[s.saveStatus] || "") + '</span></div>' +

        '<div class="q-type-note">' + (isMulti ? "Select all correct answers" : "Select one answer") + '</div>' +

        '<div class="q-text">' + escapeHtml(q.text) + '</div>' +

        '<div class="options">' + optsHtml + '</div>' +

        '<div class="dots">' + dotsHtml + '</div>' +

        errBanner +

        '<div class="quiz-actions">' +

          '<button class="btn" id="prevBtn" ' + (qi===0 ? "disabled" : "") + '>&larr; Previous</button>' +

          (qi < t.questions.length - 1

            ? '<button class="btn primary" id="nextBtn">Next &rarr;</button>'

            : '<button class="btn primary" id="submitBtn">Submit (' + answeredCount + '/' + t.questions.length + ' answered)</button>') +

        '</div>' +

      '</div></div>';


    app.querySelectorAll(".option").forEach(function(btn){

      btn.onclick = function(){

        var i = parseInt(btn.getAttribute("data-opt"), 10);

        if (isMulti) {

          var arr = s.answers[q.id].slice(); var pos = arr.indexOf(i);

          if (pos === -1) arr.push(i); else arr.splice(pos, 1);

          s.answers[q.id] = arr;

        } else s.answers[q.id] = i;

        writeCache();

        s.saveStatus = "saving"; renderStudentQuiz();

        apiPost("saveAttemptProgress", { attemptId: s.attemptId, currentQuestion: s.current, answers: s.answers })

          .then(function(){ s.saveStatus = "saved"; if (currentlyOnQuiz(s.testId)) renderStudentQuiz(); })

          .catch(function(){ s.saveStatus = "offline"; if (currentlyOnQuiz(s.testId)) renderStudentQuiz(); });

      };

    });

    app.querySelectorAll(".dot").forEach(function(d){

      d.onclick = function(){ s.current = parseInt(d.getAttribute("data-jump"),10); writeCache(); renderStudentQuiz(); };

    });

    var prevBtn = $("#prevBtn"); if (prevBtn) prevBtn.onclick = function(){ s.current--; writeCache(); renderStudentQuiz(); };

    var nextBtn = $("#nextBtn"); if (nextBtn) nextBtn.onclick = function(){ s.current++; writeCache(); renderStudentQuiz(); };

    var submitBtn = $("#submitBtn");

    if (submitBtn) submitBtn.onclick = function(){

      var unanswered = t.questions.length - answeredCount;

      if (unanswered > 0 && !confirm(unanswered + " question(s) unanswered. Submit anyway?")) return;

      doSubmit();

    };

  }


  function doSubmit(){

    var s = studentSession;

    s.submitError = null;

    var btn = $("#submitBtn"); if (btn) { btn.disabled = true; btn.textContent = "Submitting…"; }

    apiPost("submitAttempt", { attemptId: s.attemptId, answers: s.answers }).then(function(res){

      s.score = res.score; s.total = res.total; s.step = "result"; writeCache(); render();

    }).catch(function(err){

      s.submitError = "Your answers are saved. We haven't submitted the test yet. (" + err.message + ")";

      if (currentlyOnQuiz(s.testId)) renderStudentQuiz();

    });

  }

  function errorLabel(d){ return d.errorType ? (d.topic + " → " + d.errorType) : d.topic; }


  function studentDetailHtml(phone){

    var subs = resultsList.filter(function(r){ return r.phone === phone; }).sort(function(a,b){ return new Date(b.submittedAt) - new Date(a.submittedAt); });

    if (subs.length === 0) { openStudentPhone = null; return submissionsTabHtml(); }

    var latest = subs[0];

    var avgPct = Math.round(subs.reduce(function(sum,r){ return sum + (r.total ? (r.score/r.total)*100 : 0); }, 0) / subs.length);


    function tally(list){

      var map = {};

      list.forEach(function(r){

        r.details.forEach(function(d){

          if (d.isCorrect) return;

          var key = errorLabel(d);

          if (!map[key]) map[key] = { wrong: 0, testIds: {} };

          map[key].wrong++; map[key].testIds[r.testId] = true;

        });

      });

      return Object.keys(map).map(function(k){ return { label: k, wrong: map[k].wrong, testCount: Object.keys(map[k].testIds).length }; })

        .sort(function(a,b){ return b.wrong - a.wrong; });

    }


    var latestRows = tally([latest]);

    var maxLatest = Math.max.apply(null, latestRows.map(function(r){ return r.wrong; }).concat([1]));

    var latestHtml = latestRows.length === 0 ? '<p class="muted small">No mistakes on this test.</p>' :

      latestRows.map(function(row){

        return '<div class="topic-bar-row"><div class="label">' + escapeHtml(row.label) + '</div>' +

          '<div class="topic-bar-track"><div class="topic-bar-fill" style="width:' + Math.round(row.wrong/maxLatest*100) + '%;background:var(--wrong);"></div></div>' +

          '<div class="pct">' + row.wrong + ' mistake(s)</div></div>';

      }).join("");


    var allRows = tally(subs);

    var maxAll = Math.max.apply(null, allRows.map(function(r){ return r.wrong; }).concat([1]));

    var allHtml = allRows.length === 0 ? '<p class="muted small">No mistakes recorded across their submissions.</p>' :

      allRows.map(function(row){

        var flag = row.testCount >= 2 ? "🔴 " : "";

        return '<div class="topic-bar-row"><div class="label">' + flag + escapeHtml(row.label) + '</div>' +

          '<div class="topic-bar-track"><div class="topic-bar-fill" style="width:' + Math.round(row.wrong/maxAll*100) + '%;background:' + (row.testCount>=2 ? "var(--wrong)" : "var(--gold)") + ';"></div></div>' +

          '<div class="pct">' + row.wrong + ' across ' + row.testCount + ' test(s)</div></div>';

      }).join("");


    var subsHtml = subs.map(function(r){

      var pct = r.total ? Math.round((r.score/r.total)*100) : 0;

      return '<div class="flag-item"><span>' + escapeHtml(r.testTitle) + ' &middot; ' + new Date(r.submittedAt).toLocaleDateString() + '</span>' +

        '<span class="n" style="color:var(--ink);">' + r.score + '/' + r.total + ' (' + pct + '%)</span></div>';

    }).join("");


    return '<div class="back-link" id="backToSubs">&larr; All submissions</div>' +

      '<div class="card"><h2>' + escapeHtml(latest.studentName) + '</h2><p class="muted small">' + escapeHtml(phone) + '</p>' +

        '<div class="row mt16">' +

          '<div class="stat-card"><div class="num">' + latest.score + '/' + latest.total + '</div><div class="lbl">Latest score</div></div>' +

          '<div class="stat-card"><div class="num">' + avgPct + '%</div><div class="lbl">Average across ' + subs.length + ' test(s)</div></div>' +

        '</div></div>' +

      '<div class="card"><h3 class="subtle-h">Errors &mdash; latest test</h3>' + latestHtml + '</div>' +

      '<div class="card"><h3 class="subtle-h">Repeated mistakes &mdash; all tests (🔴 = seen in 2+ tests)</h3>' + allHtml + '</div>' +

      '<div class="card"><h3 class="subtle-h">All submissions</h3>' + subsHtml + '</div>';

  }


  function dashboardTabHtml(){

    if (resultsList.length === 0) return '<div class="card"><div class="empty-note">No submissions yet — analytics will appear here once students start completing tests.</div></div>';

    var uniquePhones = {}; resultsList.forEach(function(r){ uniquePhones[r.phone] = true; });

    var avgPct = Math.round(resultsList.reduce(function(s,r){ return s + (r.total ? (r.score/r.total)*100 : 0); }, 0) / resultsList.length);


    var map = {};

    resultsList.forEach(function(r){

      r.details.forEach(function(d){

        if (d.isCorrect) return;

        var key = errorLabel(d);

        if (!map[key]) map[key] = { wrong: 0, testIds: {} };

        map[key].wrong++; map[key].testIds[r.testId] = true;

      });

    });

    var rows = Object.keys(map).map(function(k){ return { label: k, wrong: map[k].wrong, testCount: Object.keys(map[k].testIds).length }; })

      .sort(function(a,b){ return b.wrong - a.wrong; });

    var mostCommon = rows.length ? rows[0].label : "—";

    var maxWrong = Math.max.apply(null, rows.map(function(r){ return r.wrong; }).concat([1]));

    var chartHtml = rows.map(function(row){

      return '<div class="topic-bar-row"><div class="label">' + escapeHtml(row.label) + '</div>' +

        '<div class="topic-bar-track"><div class="topic-bar-fill" style="width:' + Math.round(row.wrong/maxWrong*100) + '%;background:var(--wrong);"></div></div>' +

        '<div class="pct">' + row.wrong + ' mistake(s)</div></div>';

    }).join("");

    var repeated = rows.filter(function(r){ return r.testCount >= 2; });

    var repeatedHtml = repeated.length === 0 ? '<p class="muted small">Nothing has recurred across multiple tests yet.</p>' :

      repeated.map(function(row){ return '<div class="flag-item">🔴 ' + escapeHtml(row.label) + '<span class="n">' + row.wrong + ' mistakes across ' + row.testCount + ' test(s)</span></div>'; }).join("");


    return '<div class="stat-grid">' +

        '<div class="stat-card"><div class="num">' + Object.keys(uniquePhones).length + '</div><div class="lbl">Total students</div></div>' +

        '<div class="stat-card"><div class="num">' + resultsList.length + '</div><div class="lbl">Submissions</div></div>' +

        '<div class="stat-card"><div class="num">' + avgPct + '%</div><div class="lbl">Average score</div></div>' +

        '<div class="stat-card"><div class="num" style="font-size:15px;">' + escapeHtml(mostCommon) + '</div><div class="lbl">Most frequent error</div></div>' +

      '</div>' +

      '<div class="card mt16"><h3 class="subtle-h">Mistakes by topic / error type &mdash; whole class</h3>' + chartHtml + '</div>' +

      '<div class="card"><h3 class="subtle-h">Repeated errors (needs additional practice)</h3>' + repeatedHtml + '</div>';

  }

  /* ---------- shared wiring for Tests / Submissions tabs ---------- */

  function wireTeacherTab(){

    if (managingTest && !questionDraft) {

      questionDraft = { id: null, topic: UNT_TOPICS[0], errorType: "", difficulty: "", type: "single", text: "", options: ["",""], correct: 0 };

    }

    var newTestBtn = $("#newTestBtn");

    if (newTestBtn) newTestBtn.onclick = function(){

      apiPost("createOrUpdateTest", { pin: pinCache, test: { title: "Untitled test", questions: [] } }).then(function(res){

        managingTest = res.test; questionDraft = null;

        $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab();

      }).catch(handleTeacherApiError);

    };

    app.querySelectorAll("[data-manage]").forEach(function(b){ b.onclick = function(){ loadTestFull(b.getAttribute("data-manage")); }; });

    app.querySelectorAll("[data-deltest]").forEach(function(b){

      b.onclick = function(){

        if (!confirm("Delete this test? Submissions already recorded for it are kept.")) return;

        apiPost("deleteTest", { pin: pinCache, testId: b.getAttribute("data-deltest") })

          .then(function(){ testsList = testsList.filter(function(t){ return t.id !== b.getAttribute("data-deltest"); }); $("#dashContent").innerHTML = testsTabHtml(); wireTeacherTab(); })

          .catch(handleTeacherApiError);

      };

    });

    app.querySelectorAll("[data-getlink]").forEach(function(b){

      b.onclick = function(){

        var id = b.getAttribute("data-getlink");

        var link = location.origin + location.pathname + "#test=" + id;

        var card = $("#linkOutputCard"); card.style.display = "block";

        card.innerHTML = '<h3>Student link</h3><div class="code-box mt8">' + escapeHtml(link) + '</div>' +

          '<button class="btn primary small mt8" id="copyLinkBtn2">Copy link</button>';

        $("#copyLinkBtn2").onclick = function(){ copyText(link); };

        card.scrollIntoView({behavior:"smooth"});

      };

    });


    var backToTests = $("#backToTests");

    if (backToTests) backToTests.onclick = function(){ managingTest = null; questionDraft = null; renderTeacher(); };

    var saveTitleBtn = $("#saveTitleBtn");

    if (saveTitleBtn) saveTitleBtn.onclick = function(){

      managingTest.title = $("#testTitleInput").value.trim() || "Untitled test";

      saveManagingTest();

    };

    app.querySelectorAll("[data-editq]").forEach(function(b){

      b.onclick = function(){ questionDraft = JSON.parse(JSON.stringify(managingTest.questions.find(function(x){ return x.id === b.getAttribute("data-editq"); }))); $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab(); };

    });

    app.querySelectorAll("[data-delq]").forEach(function(b){

      b.onclick = function(){

        if (!confirm("Delete this question?")) return;

        managingTest.questions = managingTest.questions.filter(function(x){ return x.id !== b.getAttribute("data-delq"); });

        saveManagingTest();

      };

    });

    var qTypeSel = $("#qType");

    if (qTypeSel) qTypeSel.onchange = function(){ syncDraftFromForm(); questionDraft.type = qTypeSel.value; questionDraft.correct = questionDraft.type === "multiple" ? [] : 0; $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab(); };

    var addOptBtn = $("#addOptBtn");

    if (addOptBtn) addOptBtn.onclick = function(){ syncDraftFromForm(); if (questionDraft.options.length < 6) questionDraft.options.push(""); $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab(); };

    app.querySelectorAll("[data-rmopt]").forEach(function(b){

      b.onclick = function(){

        syncDraftFromForm();

        var i = parseInt(b.getAttribute("data-rmopt"), 10);

        questionDraft.options.splice(i, 1);

        if (questionDraft.type === "multiple") questionDraft.correct = questionDraft.correct.filter(function(c){ return c !== i; }).map(function(c){ return c > i ? c-1 : c; });

        else if (questionDraft.correct === i) questionDraft.correct = 0; else if (questionDraft.correct > i) questionDraft.correct--;

        $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab();

      };

    });

    var saveQBtn = $("#saveQBtn");

    if (saveQBtn) saveQBtn.onclick = function(){

      syncDraftFromForm();

      var topic = $("#qTopic").value, errorType = $("#qErrorType").value.trim(), difficulty = $("#qDifficulty").value.trim(), text = $("#qText").value.trim();

      if (!text) { toast("Enter the question text"); return; }

      if (questionDraft.options.some(function(o){ return !o.trim(); })) { toast("Fill in every option, or remove empty ones"); return; }

      if (questionDraft.type === "single" && questionDraft.correct == null) { toast("Mark the correct answer"); return; }

      if (questionDraft.type === "multiple" && questionDraft.correct.length === 0) { toast("Mark at least one correct answer"); return; }

      var finalQ = { id: questionDraft.id || uid(), topic: topic, errorType: errorType, difficulty: difficulty, type: questionDraft.type, text: text, options: questionDraft.options, correct: questionDraft.correct };

      if (questionDraft.id) { var idx = managingTest.questions.findIndex(function(x){ return x.id === questionDraft.id; }); managingTest.questions[idx] = finalQ; }

      else managingTest.questions.push(finalQ);

      questionDraft = null; saveManagingTest();

    };

    var cancelEditBtn = $("#cancelEditBtn");

    if (cancelEditBtn) cancelEditBtn.onclick = function(){ questionDraft = null; $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab(); };

    app.querySelectorAll("[data-optfield]").forEach(function(el){ el.oninput = function(){ questionDraft.options[parseInt(el.getAttribute("data-optfield"),10)] = el.value; }; });

    app.querySelectorAll("[data-correct]").forEach(function(el){ el.onchange = function(){ syncDraftFromForm(); }; });


    app.querySelectorAll("[data-openphone]").forEach(function(row){ row.onclick = function(){ openStudentPhone = row.getAttribute("data-openphone"); $("#dashContent").innerHTML = submissionsTabHtml(); wireTeacherTab(); }; });

    var backToSubs = $("#backToSubs");

    if (backToSubs) backToSubs.onclick = function(){ openStudentPhone = null; $("#dashContent").innerHTML = submissionsTabHtml(); wireTeacherTab(); };

  }


  function saveManagingTest(){

    apiPost("createOrUpdateTest", { pin: pinCache, test: managingTest }).then(function(res){

      managingTest = res.test; toast("Saved"); $("#dashContent").innerHTML = questionEditorHtml(); wireTeacherTab();

    }).catch(handleTeacherApiError);

  }


  function syncDraftFromForm(){

    if (!questionDraft) return;

    app.querySelectorAll("[data-optfield]").forEach(function(el){ questionDraft.options[parseInt(el.getAttribute("data-optfield"),10)] = el.value; });

    if (questionDraft.type === "multiple") {

      var arr = []; app.querySelectorAll('[data-correct]').forEach(function(c){ if (c.checked) arr.push(parseInt(c.getAttribute("data-correct"),10)); });

      questionDraft.correct = arr;

    } else {

      app.querySelectorAll('[data-correct]').forEach(function(r){ if (r.checked) questionDraft.correct = parseInt(r.getAttribute("data-correct"),10); });

    }

  }

})();
